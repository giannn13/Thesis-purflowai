//data flow level description of the logic diagram in Figure 7.1
module Comb_Ckt1_DF(input x, y, z, output F, G);
	wire k;

	assign k = y ^ z;
	assign G = k ^ x;
	assign F = (~k & z) | (k & x);

endmodule
